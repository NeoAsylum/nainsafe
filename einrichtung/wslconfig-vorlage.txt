# Konfiguration der WSL2-VM, die die Agentenfabrik traegt.
#
# Angelegt am 2026-09-05, nachdem die VM gegen 20:20 verschwunden war und die Fabrik
# 74 Minuten stillstand. Vorzeichen war ein `Wsl/0x8007000e` ("nicht genuegend
# Speicherressourcen") vier Minuten zuvor -- der Druck lag auf der Windows-Seite.
#
# systemd laeuft in der Distro, die VM hatte also durchgehend Prozesse. Es war kein
# Leerlauf-Abschalten, sondern Speicherdruck: WSL nahm sich voreingestellt die Haelfte
# des Wirtsspeichers (16 von 32 GB) und gab sie nicht zurueck, auch wenn die Fabrik nur
# ein Gigabyte brauchte.

[wsl2]

# Ein Deckel statt der Haelfte -- der wirksame Schalter gegen den Druck auf den Wirt.
# Gemessen am 2026-09-05: acht gleichzeitige Agenten plus Compiler brauchten unter einem
# Gigabyte; zehn sind reichlich und lassen dem Wirt zweiundzwanzig.
memory=10GB

# Acht von zwoelf Kernen. Die Fabrik faehrt acht Pakete gleichzeitig; die uebrigen vier
# bleiben fuer Windows, damit der Rechner unter Vollast bedienbar bleibt.
processors=8

swap=4GB

# ---------------------------------------------------------------------------
# Zwei Schluessel, die hier standen und wieder heraus sind -- beide gemessen:
#
# `autoMemoryReclaim=gradual` -- WSL 2.7.13 kennt ihn nicht und warnt bei **jedem**
# Aufruf darueber. Eine Warnung, die man nach dem dritten Mal ueberliest, verdeckt die
# naechste echte.
#
# `vmIdleTimeout=-1` -- sollte heissen "nie abschalten" und bewirkte das Gegenteil. Mit
# ihm startete die VM bei fast jedem `wsl.exe`-Aufruf neu (Startup finished um 21:50:36
# und 21:51:15, keine Minute auseinander), und ein im Hintergrund gestarteter Tageslauf
# starb eine Minute nach dem Start. Ohne ihn -- also mit der Voreinstellung -- haben
# Hintergrundlaeufe am selben Tag nachweislich 1h50 und 1h24 durchgehalten.
#
# Die Lehre gilt fuer beide: Ein Schalter, der plausibel klingt, ist keine Messung. Der
# Beleg, dass die VM anbleibt, ist ein Hintergrundprozess, der einen Client-Abbruch
# ueberlebt -- nicht die Zeile in dieser Datei.
# ---------------------------------------------------------------------------
